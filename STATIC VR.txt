#ifndef M_PI
#define M_PI 3.14159265358979323846
#endif

void fEllipse(int c, int x, int y, int rx, int ry) {
    setcolor(c); setfillstyle(SOLID_FILL, c); fillellipse(x, y, rx, ry);
}

void fPoly(int c, int n, int* p) {
    setcolor(c); setfillstyle(SOLID_FILL, c); fillpoly(n, p);
}

struct Star { double x, y, driftSpeed; int baseBrightness; double twinklePhase; };

// œÊ«· «·—”„
// 1.  «· ÊÂÃ Œ·› «·‘„”
void drawSkyGlow(int sunX, int sunY, double darkness) {
    for(int r = 400; r > 50; r -= 50) {
        int c = (int)(40 * ((400.0 - r) / 350.0) * (1.0 - darkness));
        if (c > 0) fEllipse(COLOR(c, c, c + 10), sunX, sunY, r, r);
    }
}

// 2. —”„ «·‰ÃÊ„ ( «·√Ê·)
void drawStars(Star* s, int count, int frame, double darkness, int screenW) {
    for (int i = 0; i < count; i++) {
        int t = s[i].baseBrightness + (int)(40 * sin(s[i].twinklePhase + frame * 0.05));
        t = t > 255 ? 255 : (t < 20 ? 20 : t);
        int b = t + (int)((255 - t) * darkness);
        putpixel((int)s[i].x, (int)s[i].y, COLOR(b, b, b));
    }
}

// 3. —”„ «·‘„” Ê√‘⁄ Â« ( «·À«‰Ì)
void drawSun(int sunX, int sunY, int radius) {
    setcolor(YELLOW);
    circle(sunX, sunY, radius);
    setfillstyle(SOLID_FILL, YELLOW);
    floodfill(sunX, sunY, YELLOW);

    int sunRay = 12;
    int maxiumRayLength = 30;
    double angleOfSun = 0.03;

    for (int i = 0; i < sunRay; i++) {
        double angle = i * (2 * M_PI / sunRay);
        for (int j = 0; j < 3; j++) {
            double factor = 1.0 - j * 0.3;
            int rayLength = maxiumRayLength * factor;
            int x1 = sunX + radius * cos(angle);
            int y1 = sunY + radius * sin(angle);
            int x2 = sunX + (radius + rayLength) * cos(angle + angleOfSun * factor);
            int y2 = sunY + (radius + rayLength) * sin(angle + angleOfSun * factor);
            int x3 = sunX + (radius + rayLength) * cos(angle - angleOfSun * factor);
            int y3 = sunY + (radius + rayLength) * sin(angle - angleOfSun * factor);
            int poly[6] = {x1, y1, x2, y2, x3, y3};
            setfillstyle(SOLID_FILL, YELLOW);
            fillpoly(3, poly);
        }
    }
}

// 4. —”„ «·ﬁ„— ( «·À«·À)
void drawMoon(int mX, int mY, int r) {
    setcolor(DARKGRAY);
    setfillstyle(SOLID_FILL, DARKGRAY);

}

// 6. —”„ «·’«—ÊŒ ( «·Œ«„”)
void drawRocket(int rX, int rY, int f, float sc = 1.8f) {
    auto sx = [&](double v) { return rX + (int)(v * sc); };
    auto sy = [&](double v) { return rY + (int)(v * sc); };

    int len = 15 + (int)(10 * sin(f * 0.8));
    int flame[] = { sx(0), sy(4), sx(-len - 5), sy(10), sx(0), sy(16) };
    fPoly((f % 4 < 2) ? COLOR(255, 140, 0) : COLOR(255, 255, 100), 3, flame);

    int core[] = { sx(0), sy(7), sx(-len/2), sy(10), sx(0), sy(13) };
    fPoly(WHITE, 3, core);

    int top[] = { sx(5), sy(0), sx(15), sy(-8), sx(25), sy(0) };
    int bot[] = { sx(5), sy(20), sx(15), sy(28), sx(25), sy(20) };
    fPoly(COLOR(150, 150, 150), 3, top); fPoly(COLOR(150, 150, 150), 3, bot);

    int bCol = COLOR(220, 220, 220); setcolor(bCol); setfillstyle(SOLID_FILL, bCol);
    bar(sx(0), sy(0), sx(60), sy(20));

    setcolor(COLOR(130, 130, 130)); line(sx(0), sy(19), sx(60), sy(19)); line(sx(0), sy(18), sx(60), sy(18));

    fEllipse(COLOR(60, 120, 220), sx(40), sy(10), (int)(6 * sc), (int)(6 * sc));
    putpixel(sx(41), sy(8), WHITE); putpixel(sx(42), sy(9), WHITE);

    int n[] = { sx(60), sy(0), sx(85), sy(10), sx(60), sy(20) };
    fPoly(COLOR(200, 30, 30), 3, n);
}

        stars[i] = { (double)(rand()%w), (double)(rand()%h), (i < 200 ? 0.4 : 0.1) + ((rand()%10)/(i < 200 ? 10.0 : 30.0)), (i < 200 ? 140 : 50) + (rand()%(i < 200 ? 60 : 50)), (rand()%100)/10.0 };
    }


    int sX = w * 0.2;
    int sY = h / 2;
    int sR = 50;

    int mX = sX + 110;      // «·ﬁ„—
    int mY = sY - 140;

    int rX = w * 0.45;      // «·’«—ÊŒ
    int rY = 140;

    int eX = w * 0.8;       // «·√—÷
    int eY = h * 0.45;
    int eR = 70;

    int frame = 20;
    double dF = 0.5;        // «· ÊÂÃ

    // 1. «·Œ·›Ì… «·”Êœ«¡
    setbkcolor(BLACK); cleardevice();

    drawSkyGlow(sX, sY, dF);
    drawStars(stars, 400, frame, dF, w);

    // 6. —”„ «·‘Â«» «·Œ›Ì› («·Œ«„”)
    setcolor(COLOR(100, 100, 100));
    line(sX + 250, sY - 30, sX + 200, sY - 70);

    drawSun(sX, sY, sR);
    drawMoon(mX, mY, sR + 2);
    drawEarth(eX, eY, eR);
    drawRocket(rX, rY, frame);

    //  ‰’  Ê÷ÌÕÌ

}
